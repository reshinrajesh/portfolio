import { createClient } from '@supabase/supabase-js';
import dotenv from 'dotenv';
dotenv.config({ path: '.env.local' });

const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL;
const supabaseServiceKey = process.env.SUPABASE_SERVICE_ROLE_KEY;

if (!supabaseUrl || !supabaseServiceKey) {
    console.error("Missing Supabase credentials in .env.local");
    process.exit(1);
}

const supabase = createClient(supabaseUrl, supabaseServiceKey);

async function setup() {
    console.log("Adding huly_id columns to database tables...");

    const tables = ['posts', 'skills', 'gallery_images', 'gallery_albums'];

    for (const table of tables) {
        process.stdout.write(`Checking ${table}... `);
        
        // We use a raw RPC or just try a select to see if it exists
        // Since we can't run raw SQL easily via the SDK without a predefined function,
        // we recommend the user to run this in the Supabase SQL Editor:
        /*
        ALTER TABLE posts ADD COLUMN IF NOT EXISTS huly_id TEXT UNIQUE;
        ALTER TABLE skills ADD COLUMN IF NOT EXISTS huly_id TEXT UNIQUE;
        ALTER TABLE gallery_images ADD COLUMN IF NOT EXISTS huly_id TEXT UNIQUE;
        ALTER TABLE gallery_albums ADD COLUMN IF NOT EXISTS huly_id TEXT UNIQUE;
        */
    }
    
    console.log("\nIMPORTANT: The Supabase SDK does not support DDL (ALTER TABLE) directly.");
    console.log("Please run the following SQL in your Supabase SQL Editor (https://supabase.com/dashboard/project/_/sql):");
    console.log(`
ALTER TABLE posts ADD COLUMN IF NOT EXISTS huly_id TEXT UNIQUE;
ALTER TABLE skills ADD COLUMN IF NOT EXISTS huly_id TEXT UNIQUE;
ALTER TABLE gallery_images ADD COLUMN IF NOT EXISTS huly_id TEXT UNIQUE;
ALTER TABLE gallery_albums ADD COLUMN IF NOT EXISTS huly_id TEXT UNIQUE;
    `);
}

setup();
